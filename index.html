@Override public void run() { log.info("WhatsApp retry scheduler triggered."); try { List> retryRecords = WhatsAppManager.fetchRetryRecordsWhatsAppWorkflow(); log.info("Records found for retry: " + retryRecords.size()); for (Map record : retryRecords) { String originalRrn = (String) record.get("ORIGINAL_RRN"); String payload = (String) record.get("PAYLOAD"); int retryCount = (int) record.get("RETRY_COUNT"); int maxRetry = (int) record.get("MAX_RETRY"); int attemptNumber = retryCount + 1; // New REQUEST_ID for this retry — EIS expects new one String newRequestId = Notification_Utility.generateReferenceNumber( NE_WhatsAppQLauncher.inputSourceId); try { log.info("Retrying EIS. originalRrn=" + originalRrn + " newRequestId=" + newRequestId + " attempt=" + attemptNumber); // Rebuild datamap from stored payload Gson gson = new Gson(); Map datamap = gson.fromJson(payload, Map.class); // Call EIS with new requestId String key = RandomStringUtils.randomAlphanumeric(32); String encryptedData = Notification_Utility .prepareEncryptedEisJsonReqforWhatsApp( datamap, key, NE_WhatsAppQLauncher.inputSourceId, newRequestId); // pass new requestId String eisUrl = NE_WhatsAppQLauncher.eisWhatsAppDns + NE_WhatsAppQLauncher.eisWhatsAppUrl; Map apiRespMap = ApiClientWhatsApp.callEISApi(encryptedData, eisUrl, key); Map eisResponse = (Map) apiRespMap.get("EIS_RESPONSE"); String statusCode = (String) eisResponse.get("STATUS_CODE"); String errorDescription = (String) eisResponse.get("DESCRIPTION"); if ("00".equals(statusCode)) { // SUCCESS WhatsAppManager.updateSuccessFlagWhatsAppWorkflow(originalRrn); WhatsAppManager.insertAuditWhatsAppWorkflow( originalRrn, newRequestId, attemptNumber, "V", null); log.info("Retry SUCCESS. originalRrn=" + originalRrn); } else { // Still failing handleFailure(originalRrn, newRequestId, attemptNumber, retryCount, maxRetry, errorDescription); } } catch (Exception e) { log.error("Retry exception. originalRrn=" + originalRrn, e); handleFailure(originalRrn, newRequestId, attemptNumber, retryCount, maxRetry, e.getMessage()); } } } catch (Exception e) { log.error("WhatsApp retry scheduler error.", e); } } private void handleFailure(String originalRrn, String currentRequestId, int attemptNumber, int retryCount, int maxRetry, String errorDetail) { try { // Audit this attempt WhatsAppManager.insertAuditWhatsAppWorkflow( originalRrn, currentRequestId, attemptNumber, "F", errorDetail); if ((retryCount + 1) >= maxRetry) { // Max retries exhausted — DLQ WhatsAppManager.updateDlqFlagWhatsAppWorkflow(originalRrn, errorDetail); log.warn("Max retries exhausted. DLQ. originalRrn=" + originalRrn); } else { // Generate next REQUEST_ID for next retry String nextRequestId = Notification_Utility.generateReferenceNumber( NE_WhatsAppQLauncher.inputSourceId); WhatsAppManager.updateFailFlagWhatsAppWorkflow( originalRrn, nextRequestId, errorDetail, retryIntervalMinutes); log.warn("Retry failed. Will retry again. originalRrn=" + originalRrn); } } catch (Exception e) { log.error("handleFailure error. originalRrn=" + originalRrn, e); } }
