package com.example.Universal.Service; import java.io.BufferedWriter; import java.io.File; import java.io.FileWriter; import java.io.IOException; import java.util.Map; import java.util.stream.Collectors; import org.springframework.stereotype.Service; import tools.jackson.databind.ObjectMapper; import tools.jackson.databind.exc.JsonNodeException; @Service public class CommonProcessingService { public void process(String channelCode, String channelIdentifier, Object validatedData) { try { // Convert object to pipe-separated TXT String txtData = convertToPipeSeparated(validatedData); // Generate file path String fileName = channelCode + "_" + System.currentTimeMillis() + "_" + java.util.UUID.randomUUID() + ".txt"; String filePath = "output/" + channelIdentifier + "/" + fileName; File file = new File(filePath); file.getParentFile().mkdirs(); // Write using BufferedWriter try (BufferedWriter writer = new BufferedWriter(new FileWriter(file))) { writer.write(txtData); } // Audit logging System.out.println("TXT generated for " + channelIdentifier + " at " + filePath); } catch (IOException e) { throw new RuntimeException("Error generating TXT file", e); } } private String convertToPipeSeparated(Object dto) throws JsonNodeException { ObjectMapper mapper = new ObjectMapper(); Map map = mapper.convertValue(dto, Map.class); return map.values().stream() .map(String::valueOf) .collect(Collectors.joining("|")); } } import java.io.*; import java.nio.file.*; import java.util.regex.*; public class MurUpdater { public static void main(String[] args) throws IOException { String folderPath = "C:/mtfiles"; // your folder Files.list(Paths.get(folderPath)) .filter(Files::isRegularFile) .forEach(path -> processFile(path)); } private static void processFile(Path path) { try { String content = new String(Files.readAllBytes(path)); Pattern pattern = Pattern.compile("\\{3:\\{108:(.*?)}}"); Matcher matcher = pattern.matcher(content); if (matcher.find()) { String oldMur = matcher.group(1); String newMur = incrementMur(oldMur); content = content.replace(oldMur, newMur); Files.write(path, content.getBytes()); System.out.println("Updated: " + path.getFileName()); } } catch (Exception e) { e.printStackTrace(); } } private static String incrementMur(String mur) { // Separate prefix and number String prefix = mur.replaceAll("\\d+$", ""); String numberPart = mur.substring(prefix.length()); long num = Long.parseLong(numberPart); num++; String newNumber = String.format("%0" + numberPart.length() + "d", num); String result = prefix + newNumber; // Ensure length = 16 if (result.length() > 16) { result = result.substring(0, 16); } else if (result.length() < 16) { result = String.format("%-16s", result).replace(' ', '0'); } return result; } }
