SELECT r.REQUEST_ID, r.MUR_CODE AS REQUEST_MUR_CODE, r.SOURCE_SYSTEM, r.MESSAGE_FORMAT, r.CREATION_TIME, c.RESPONSE_ID, c.MUR_CODE AS RESPONSE_MUR_CODE, c.RESPONSE_TIME, c.RESPONSE_CODE, c.RESPONSE_STATUS FROM ANC_SWIFT_REQUEST r FULL OUTER JOIN ANC_SWIFT_CSIG_RESPONSE c ON r.MUR_CODE = c.MUR_CODE WHERE TRUNC(c.RESPONSE_TIME) = TRUNC(SYSDATE) ORDER BY c.RESPONSE_TIME DESC;
