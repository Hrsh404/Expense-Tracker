CREATE TABLE WHATSAPP_WORKFLOW ( ID NUMBER UNIQUE, ORIGINAL_REQUEST_ID VARCHAR2(50), ALERT_CODE VARCHAR2(20), PAYLOAD CLOB, STATUS VARCHAR2(20), RETRY_COUNT NUMBER DEFAULT 0, MAX_RETRY NUMBER DEFAULT 3, NEXT_RETRY_AT TIMESTAMP(6) DEFAULT SYSTIMESTAMP, CREATED_AT TIMESTAMP(6) DEFAULT SYSTIMESTAMP, LAST_UPDATED_AT TIMESTAMP(6) DEFAULT SYSTIMESTAMP, ERROR_CODE VARCHAR2(20), ERROR_DETAIL VARCHAR2(200), CLAIMED_BY VARCHAR2(100) ); CREATE SEQUENCE WHATSAPP_WORKFLOW_SEQUENCE START WITH 1 INCREMENT BY 1; CREATE INDEX IDX_WHATSAPP_STATUS ON WHATSAPP_WORKFLOW(STATUS); CREATE INDEX IDX_WHATSAPP_NEXT_RETRY ON WHATSAPP_WORKFLOW(NEXT_RETRY_AT, STATUS);
