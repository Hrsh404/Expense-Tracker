package com.sbi.ittss.swift.client.service; import java.util.List; import org.springframework.context.annotation.AnnotationConfigApplicationContext; public class ResponseServiceDebug { public static void main(String[] args) { AnnotationConfigApplicationContext context = new AnnotationConfigApplicationContext(); context.scan("com.sbi.ittss"); context.refresh(); try { ResponseService responseService = context.getBean(ResponseService.class); System.out.println("Starting CSIG response processing..."); List responses = responseService.processAllCsigResponses(); System.out.println("Processing completed."); if (responses != null) { System.out.println( "Responses processed: " + responses.size()); for (int i = 0; i < responses.size(); i++) { CsigResponse response = responses.get(i); System.out.println( "Response " + (i + 1) + " | MUR: " + response.getMurCode() + " | Code: " + response.getResponseCode()); } } else { System.out.println("Returned response list is null."); } } catch (Exception e) { System.err.println("Processing failed!"); e.printStackTrace(); } finally { context.close(); } } }
