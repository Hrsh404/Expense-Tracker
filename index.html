import com.github.javaparser.JavaParser; import com.github.javaparser.ParseResult; import com.github.javaparser.ParserConfiguration; import com.github.javaparser.ast.CompilationUnit; import com.github.javaparser.ast.Node; import com.github.javaparser.ast.NodeList; import com.github.javaparser.ast.body.VariableDeclarator; import com.github.javaparser.ast.expr.Expression; import com.github.javaparser.ast.expr.MethodCallExpr; import com.github.javaparser.ast.stmt.BlockStmt; import com.github.javaparser.ast.stmt.ExpressionStmt; import com.github.javaparser.ast.stmt.Statement; import com.github.javaparser.ast.stmt.TryStmt; import com.github.javaparser.ast.visitor.ModifierVisitor; import com.github.javaparser.ast.visitor.Visitable; import com.github.javaparser.ast.type.ClassOrInterfaceType; import com.github.javaparser.ast.body.Parameter; import com.github.javaparser.ast.stmt.CatchClause; import com.github.javaparser.ast.comments.Comment; import com.github.javaparser.ast.ImportDeclaration; import java.io.IOException; import java.nio.charset.StandardCharsets; import java.nio.file.*; import java.util.ArrayList; import java.util.List; import java.util.Locale; import java.util.regex.Pattern; import java.util.stream.Stream; public class CommentRemover { private static final Pattern PRESERVED_CR = Pattern.compile("\\bCR\\b", Pattern.CASE_INSENSITIVE); private static final Pattern IMPORT_LINE = Pattern.compile("(?m)^\\s*import\\s+[^;]+;\\s*"); private static final Pattern JAVA_IDENTIFIER = Pattern.compile("\\b[A-Za-z_$][A-Za-z0-9_$]*\\b"); public static void main(String[] args) { try { java.util.Scanner scanner = new java.util.Scanner(System.in); System.out.print("Enter workspace root folder: "); Path input = Paths.get(scanner.nextLine().trim()) .toAbsolutePath().normalize(); System.out.print("Enter output workspace folder: "); Path output = Paths.get(scanner.nextLine().trim()) .toAbsolutePath().normalize(); scanner.close(); if (!Files.isDirectory(input)) { System.out.println( "Input must be an existing workspace directory."); return; } if (input.equals(output) || input.startsWith(output) || output.startsWith(input)) { System.out.println( "Input and output folders must be separate."); return; } processWorkspace(input, output); System.out.println("Workspace processing completed."); } catch (IOException e) { System.out.println("File error: " + e.getMessage()); } catch (InvalidPathException e) { System.out.println("Invalid path: " + e.getMessage()); } } private static void processWorkspace( Path input, Path output) throws IOException { Files.createDirectories(output); try (Stream paths = Files.walk(input)) { paths.forEach(source -> { try { Path relative = input.relativize(source); Path destination = output.resolve(relative); if (Files.isDirectory(source)) { Files.createDirectories(destination); return; } Files.createDirectories(destination.getParent()); if (source.getFileName().toString() .endsWith(".java")) { String code = Files.readString( source, StandardCharsets.UTF_8); String cleaned = cleanJavaCode(code); Files.writeString( destination, cleaned, StandardCharsets.UTF_8); System.out.println("Processed: " + relative); } else { Files.copy( source, destination, StandardCopyOption.REPLACE_EXISTING); } } catch (IOException e) { throw new java.io.UncheckedIOException(e); } }); } catch (java.io.UncheckedIOException e) { throw e.getCause(); } } private static String cleanJavaCode(String code) { ParserConfiguration configuration = new ParserConfiguration(); configuration.setLanguageLevel( ParserConfiguration.LanguageLevel.BLEEDING_EDGE); JavaParser parser = new JavaParser(configuration); ParseResult parsed = parser.parse(code); if (!parsed.isSuccessful() || !parsed.getResult().isPresent()) { throw new IllegalArgumentException( "Unable to parse Java source. Original source must " + "be reviewed before transformation."); } CompilationUnit cu = parsed.getResult().get(); // Remove comments except comments containing CR in any case. for (Comment comment : new ArrayList<>(cu.getAllComments())) { if (!PRESERVED_CR.matcher(comment.getContent()).find()) { comment.remove(); } } // Remove logging statements and printStackTrace calls. cu.accept(new ModifierVisitor() { @Override public Visitable visit( ExpressionStmt statement, Void arg) { Expression expression = statement.getExpression(); if (expression.isMethodCallExpr() && shouldRemoveCall( expression.asMethodCallExpr())) { return null; } return super.visit(statement, arg); } }, null); // Every try block receives exactly two catch clauses. // Existing catch bodies are replaced with message extraction. cu.accept(new ModifierVisitor() { @Override public Visitable visit(TryStmt tryStmt, Void arg) { super.visit(tryStmt, arg); NodeList catches = new NodeList<>(); catches.add(createCatch("IllegalArgumentException")); catches.add(createCatch("Exception")); tryStmt.setCatchClauses(catches); return tryStmt; } }, null); // Remove apparently unused explicit imports. removeUnusedImports(cu); return cu.toString(); } private static CatchClause createCatch(String exceptionType) { Parameter parameter = new Parameter( new ClassOrInterfaceType(null, exceptionType), "e"); BlockStmt body = new BlockStmt(); VariableDeclarator errorMessage = new VariableDeclarator( new ClassOrInterfaceType(null, "String"), "errorMessage", new MethodCallExpr( new com.github.javaparser.ast.expr.NameExpr("e"), "getMessage")); body.addStatement( new com.github.javaparser.ast.stmt.ExpressionStmt( new com.github.javaparser.ast.expr.VariableDeclarationExpr( errorMessage))); return new CatchClause(parameter, body); } private static boolean shouldRemoveCall(MethodCallExpr call) { String expression = call.toString(); String methodName = call.getNameAsString(); // Remove printStackTrace() calls. if (methodName.equals("printStackTrace")) { return true; } // Remove System.out and System.err output calls. if (expression.matches( "(?s)System\\s*\\.\\s*(out|err)\\s*\\.\\s*" + "(print|println|printf)\\s*\\(.*")) { return true; } // Remove common logging framework calls at every level. if (call.getScope().isPresent()) { String scope = call.getScope().get().toString(); String lowerMethod = methodName.toLowerCase(Locale.ROOT); boolean loggingLevel = lowerMethod.matches( "trace|debug|info|warn|error|fatal|" + "log|logp|severe|warning|fine|finer|finest"); boolean loggerScope = scope.matches( "(?s).*(logger|log|log4j|slf4j).*"); if (loggingLevel && loggerScope) { return true; } } // Remove empty print calls, including print(""). if (call.getScope().isPresent() && call.getScope().get().toString() .matches("System\\s*\\.\\s*out") && (methodName.equals("print") || methodName.equals("println")) && (call.getArguments().isEmpty() || call.getArguments().size() == 1 && call.getArgument(0).isStringLiteralExpr() && call.getArgument(0) .asStringLiteralExpr() .asString().trim().isEmpty())) { return true; } return false; } private static void removeUnusedImports(CompilationUnit cu) { List imports = new ArrayList<>(cu.getImports()); for (ImportDeclaration declaration : imports) { if (declaration.isAsterisk()) { continue; } String importedName = declaration.getName().getIdentifier(); if (declaration.isStatic()) { // Static imports need more careful resolution. continue; } String sourceWithoutImports = IMPORT_LINE .matcher(cu.toString()) .replaceAll(""); boolean used = JAVA_IDENTIFIER .matcher(sourceWithoutImports) .results() .anyMatch(match -> match.group().equals(importedName)); if (!used) { declaration.remove(); } } } }
